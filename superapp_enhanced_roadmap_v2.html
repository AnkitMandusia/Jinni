# FastAPI dependency — role guard
def require_role(*roles: str):
    def checker(current_user = Depends(get_current_user)):
        if current_user.role not in roles:
            raise HTTPException(403, "Access denied for this role")
        if current_user.status != "active":
            raise HTTPException(403, f"Account {current_user.status}")
        return current_user
    return checker

Grocery order dispatch flow

Race-to-accept model — jo pehle accept kare uska order, baaki ko cancel notification

Complete flow: Customer → Stores → Delivery → Customer

1
Customer places order
POST /grocery/orders → inventory reserved (reserved_qty++) → order status = pending_dispatch
2
Nearest stores ko broadcast
Backend: radius ke andar sabhi active stores dhundho (PostGIS ya haversine formula) → sabhi ko simultaneously Firebase FCM notification bhejo with accept_timeout = 45 seconds
3
Store accepts (race condition handled)
POST /orders/{id}/store-accept → Redis distributed lock use karo → pehla accept winner → order status = store_accepted, store_id set karo
4
Baaki stores ko cancel notification
FCM multicast → "Yeh order kisi aur ne le liya" → unke app se accept button disappear
5
Customer ko confirmation
FCM push → "Order confirm! [Store Name] prepare kar raha hai" + estimated time
6
Delivery agent assign
Nearest available agent dhundo → FCM notification → agent accepts → customer + admin ko update
7
Live tracking
Delivery agent app → POST /delivery/location (every 10s) → Redis mein store → customer + admin WebSocket se live dekh sakte hain
8
Order delivered
Agent marks delivered → OTP confirm ya photo proof → order = delivered → payment settle → rating prompt
Timeout handling: Agar 45 seconds mein koi store accept na kare → auto-expand radius → phir broadcast. 3 attempts ke baad → order = no_store_available → customer ko notification + auto-refund.

Race condition fix — Redis distributed lock

# app/modules/grocery/service.py
import redis
import uuid

async def accept_order_by_store(order_id: str, store_id: str, redis_client: Redis):
    lock_key = f"order_accept_lock:{order_id}"
    lock_value = str(uuid.uuid4())

    # Try to acquire lock (SET NX EX) — atomic
    acquired = await redis_client.set(lock_key, lock_value, nx=True, ex=10)

    if not acquired:
        raise HTTPException(409, "Order already accepted by another store")

    try:
        # Double-check in DB (still pending?)
        order = await db.get(Order, order_id)
        if order.status != 'pending_dispatch':
            raise HTTPException(409, "Order no longer available")

        # Assign store
        order.store_id = store_id
        order.status = 'store_accepted'
        await db.commit()

        # Notify other stores: cancel their pending notification
        await cancel_other_store_notifications(order_id, store_id)
        return order
    finally:
        # Release lock only if we own it
        if await redis_client.get(lock_key) == lock_value:
            await redis_client.delete(lock_key)

# FastAPI permission checker
def can_edit_product(user: User, product: Product, field: str) -> bool:
    if user.role == 'admin':
        return True
    if user.role == 'store':
        # Check if admin granted this store product edit permission
        override = get_user_override(user.id, 'product', 'update')
        if not override:
            return False
        # Even if granted, only specific fields
        allowed_fields = ['is_active', 'stock_quantity']
        return field in allowed_fields
    if user.role == 'restaurant':
        # Can only edit their own menu items
        return product.restaurant_id == user.profile.restaurant_id
    return False

FCM implementation

# app/modules/notifications/service.py
from firebase_admin import messaging
from enum import Enum

class NotifType(str, Enum):
    ORDER_INCOMING    = "order_incoming"    # Store ko — actionable
    ORDER_CANCELLED   = "order_cancelled"   # Other stores ko
    AGENT_ASSIGNED    = "agent_assigned"    # Customer ko
    ORDER_DELIVERED   = "order_delivered"   # Customer ko
    PARTNER_APPROVED  = "partner_approved"  # Store/restaurant ko

async def send_to_stores_broadcast(order_id: str, nearby_store_tokens: list[str]):
    """Sab nearby stores ko ek saath bhejo"""
    message = messaging.MulticastMessage(
        tokens=nearby_store_tokens,
        data={
            "type": NotifType.ORDER_INCOMING,
            "order_id": order_id,
            "accept_url": f"/orders/{order_id}/store-accept",
            "timeout_seconds": "45",
        },
        android=messaging.AndroidConfig(
            priority="high",
            ttl=45,           # 45s ke baad FCM discard kar de
        ),
        apns=messaging.APNSConfig(
            headers={"apns-priority": "10", "apns-expiration": "45"}
        )
    )
    response = messaging.send_each_for_multicast(message)
    return response

async def cancel_pending_store_notifications(order_id: str, winner_store_id: str):
    """Baaki stores ko cancel bhejo"""
    other_tokens = get_other_store_tokens(order_id, exclude=winner_store_id)
    messaging.MulticastMessage(
        tokens=other_tokens,
        data={"type": NotifType.ORDER_CANCELLED, "order_id": order_id}
    )

Gaps identified and fixed

Pehle roadmap mein jo missing tha — sab cover kiya gaya

Gap 1: Single app, multiple roles
Pehle plan mein separate apps the implicitly. Ab: ek React Native app, role detect karo JWT se, alag navigator stack serve karo. Customer, store, delivery — sab ek APK mein, sirf view alag.
Gap 2: Store accept race condition
Multiple stores simultaneously accept karne ki koshish karenge. Fix: Redis distributed lock (SET NX EX) — atomic operation, first writer wins, guaranteed no double-accept.
Gap 3: No-accept timeout
Koi store accept na kare toh? Celery beat task: har 45s mein check karo pending_dispatch orders → expand radius ya notify customer "no store available".
Gap 4: Product edit permissions
Pehle hardcoded tha — sirf admin. Ab: user_permission_overrides table → admin kisi bhi store ko limited product edit permission de sakta hai (sirf availability/stock, price nahi).
Gap 5: Partner approval workflow
Store/restaurant/driver signup ke baad seedha active nahi hone chahiye. Fix: status = pending_approval → admin review → approve → FCM notification to partner → active.
Gap 6: Delivery OTP / proof
Delivery confirm kaise hoga? Fix: OTP-on-delivery (customer ko OTP milega, agent enter karega) ya photo proof (agent delivery photo upload kare). Both options configurable per module.
Gap 7: COD reconciliation
Cash on delivery mein cash track nahi tha. Fix: cod_collections table — agent ki collected cash, settlement status, daily reconciliation report in admin panel.
Gap 8: Admin WebSocket for live tracking
Admin ko sab orders simultaneously dekhne hain. Fix: WebSocket room "admin" → har order event, delivery agent location update push hota hai. Frontend: React + Leaflet/Google Maps embed.
Gap 9: FCM token refresh
FCM token expire ya change hota rehta hai. Fix: Mobile app mein onTokenRefresh listener → POST /users/fcm-token → DB update. Stale token detection: FCM response mein UNREGISTERED aaye toh delete karo.
